var checks = 0;
void Check(bool condition, string name)
{
    if (!condition) throw new Exception(name);
    checks++;
    Console.WriteLine("PASS: " + name);
}
void Skip(string name) => Console.WriteLine("SKIP: " + name);

bool IsValid(object model)
{
    var results = new List<ValidationResult>();
    return Validator.TryValidateObject(model, new ValidationContext(model), results, true);
}

GateEntry ValidGate(string rst) => new()
{
    RSTNumber = rst, InwardNo = "TEST-INWARD", VehicleId = 1, PartyId = 1, DriverId = 1,
    GrossWeight = 1, CategoryId = 1, ItemCategoryId = 1
};
Check(IsValid(ValidGate("1")), "RST accepts one digit");
Check(IsValid(ValidGate("123456")), "RST accepts six digits");
Check(!IsValid(ValidGate("1234567")), "RST rejects more than six digits");
Check(!IsValid(ValidGate("12A")), "RST rejects non-numeric characters");

string Read(string path) => File.ReadAllText(Path.Combine(root, path));
string? ReadOptional(string path)
{
    var fullPath = Path.Combine(root, path);
    return File.Exists(fullPath) ? File.ReadAllText(fullPath) : null;
}

int Occurrences(string source, string value) =>
    source.Split(value, StringSplitOptions.None).Length - 1;

var mergeMarkerFiles = Directory.EnumerateFiles(root, "*", SearchOption.AllDirectories)
    .Where(path => !path.Contains($"{Path.DirectorySeparatorChar}.git{Path.DirectorySeparatorChar}")
        && !path.Contains($"{Path.DirectorySeparatorChar}bin{Path.DirectorySeparatorChar}")
        && !path.Contains($"{Path.DirectorySeparatorChar}obj{Path.DirectorySeparatorChar}")
        && new[] { ".cs", ".cshtml", ".js", ".css", ".sql", ".json" }.Contains(Path.GetExtension(path), StringComparer.OrdinalIgnoreCase))
    .Where(path =>
    {
        var content = File.ReadAllText(path);
        return content.Contains("<<<<<<< ") || content.Contains(">>>>>>> ");
    })
    .ToList();
Check(mergeMarkerFiles.Count == 0, "project source contains no unresolved merge markers");

if (args.Contains("--database"))
{
    var configuration = new ConfigurationBuilder().SetBasePath(root).AddJsonFile("appsettings.json").AddEnvironmentVariables().Build();
    var connectionString = configuration.GetConnectionString("DefaultConnection") ?? throw new Exception("DefaultConnection missing");
    await using var connection = new SqlConnection(connectionString);
    await connection.OpenAsync();

    var requiredObjects = new[]
    {
        "dbo.tbl_company", "dbo.tbl_subcompany", "dbo.tbl_OrderReturn", "dbo.tbl_OrderReturnDispatch",
        "dbo.tbl_OrderReturnMovement", "dbo.tbl_OrderReturnWorker", "dbo.tbl_OrderReturnStatusHistory",
        "dbo.tbl_OrderReturnBagDetail", "dbo.sp_ManageLocationMaster", "dbo.sp_GetSupervisorDashboard",
        "dbo.sp_CreateOrderReturn", "dbo.sp_DecideOrderReturn", "dbo.sp_CreateOrderReturnDispatch",
        "dbo.sp_SaveOrderReturnMovement", "dbo.sp_SaveOrderReturnWeight", "dbo.sp_CompleteOrderReturnOutward",
        "dbo.sp_GetMethAssignments", "dbo.sp_GetMethDashboard", "dbo.sp_MethCompleteUnloadWithReturn",
        "dbo.t_SupervisorMethEntry", "dbo.t_SupervisorMethLooseEntry", "dbo.sp_GetSupervisorMethEntry",
        "dbo.sp_GetSupervisorMethLooseEntry", "dbo.sp_SaveSupervisorMethEntry"
    };
    foreach (var objectName in requiredObjects)
    {
        await using var command = new SqlCommand("SELECT OBJECT_ID(@Name)", connection);
        command.Parameters.AddWithValue("@Name", objectName);
        var objectId = await command.ExecuteScalarAsync();
        Check(objectId is not null && objectId != DBNull.Value, "database object exists: " + objectName);
    }

    await using (var rstUnique = new SqlCommand(@"
SELECT COUNT(1)
FROM sys.indexes i
JOIN sys.index_columns ic ON ic.object_id=i.object_id AND ic.index_id=i.index_id
JOIN sys.columns c ON c.object_id=ic.object_id AND c.column_id=ic.column_id
WHERE i.object_id=OBJECT_ID('dbo.t_GateEntry') AND i.is_unique=1 AND c.name='RSTNumber'", connection))
        Check(Convert.ToInt32(await rstUnique.ExecuteScalarAsync()) > 0, "database has a unique key/index on RST number");

    await using (var nullableCommand = new SqlCommand("SELECT is_nullable FROM sys.columns WHERE object_id=OBJECT_ID('dbo.m_LocationMaster') AND name='SubCompanyId'", connection))
        Check(Convert.ToBoolean(await nullableCommand.ExecuteScalarAsync()), "database allows NULL subcompany for direct locations");

    await using var transaction = (SqlTransaction)await connection.BeginTransactionAsync();
    var testName = "ROLLBACK DIRECT " + Guid.NewGuid().ToString("N")[..8];
    try
    {
        int officeId;
        await using (var officeCommand = new SqlCommand("SELECT TOP(1) OfficeId FROM dbo.tbl_company WHERE IsActive=1 AND OfficeId IS NOT NULL ORDER BY CompanyId", connection, transaction))
            officeId = Convert.ToInt32(await officeCommand.ExecuteScalarAsync());

        await using (var insert = new SqlCommand("dbo.sp_ManageLocationMaster", connection, transaction) { CommandType = System.Data.CommandType.StoredProcedure })
        {
            insert.Parameters.AddWithValue("@Action", "INSERT");
            var code = insert.Parameters.Add("@LocationCode", System.Data.SqlDbType.NVarChar, 20);
            code.Direction = System.Data.ParameterDirection.InputOutput;
            code.Value = DBNull.Value;
            insert.Parameters.AddWithValue("@LocationName", testName);
            insert.Parameters.AddWithValue("@LocationType", "Godown");
            insert.Parameters.AddWithValue("@Capacity", 1m);
            insert.Parameters.AddWithValue("@CapacityUnit", "MT");
            insert.Parameters.AddWithValue("@CreatedBy", 1);
            insert.Parameters.AddWithValue("@OfficeId", officeId);
            insert.Parameters.AddWithValue("@SubCompanyId", DBNull.Value);
            await insert.ExecuteNonQueryAsync();
        }

        await using var verify = new SqlCommand("SELECT COUNT(1) FROM dbo.m_LocationMaster l JOIN dbo.t_OfficeLocationMapping m ON m.LocationId=l.LocationId AND m.IsActive=1 WHERE l.LocationName=@Name AND l.SubCompanyId IS NULL AND m.OfficeId=@OfficeId", connection, transaction);
        verify.Parameters.AddWithValue("@Name", testName);
        verify.Parameters.AddWithValue("@OfficeId", officeId);
        Check(Convert.ToInt32(await verify.ExecuteScalarAsync()) == 1, "direct location saves with company mapping and NULL subcompany");
    }
    finally
    {
        await transaction.RollbackAsync();
    }

    await using var persisted = new SqlCommand("SELECT COUNT(1) FROM dbo.m_LocationMaster WHERE LocationName=@Name", connection);
    persisted.Parameters.AddWithValue("@Name", testName);
    Check(Convert.ToInt32(await persisted.ExecuteScalarAsync()) == 0, "direct location integration test rolls back cleanly");

    await using var methTransaction = (SqlTransaction)await connection.BeginTransactionAsync();
    try
    {
        await using var candidate = new SqlCommand(@"
SELECT TOP (1) u.UnloadId, u.SupervisorId, b.BagTypeId
FROM dbo.t_UnloadTransaction u
CROSS JOIN dbo.m_BagType b
WHERE ISNULL(u.SupervisorId,0)>0 AND ISNULL(u.MethId,0)>0 AND b.IsActive=1
ORDER BY u.UnloadId DESC, b.BagTypeId", connection, methTransaction);
        await using var candidateReader = await candidate.ExecuteReaderAsync();
        Check(await candidateReader.ReadAsync(), "working database has an eligible RST for Meth entry rollback test");
        var unloadId = candidateReader.GetInt32(0);
        var supervisorId = candidateReader.GetInt32(1);
        var bagTypeId = candidateReader.GetInt32(2);
        await candidateReader.CloseAsync();

        await using var saveMeth = new SqlCommand("dbo.sp_SaveSupervisorMethEntry", connection, methTransaction) { CommandType = System.Data.CommandType.StoredProcedure };
        saveMeth.Parameters.AddWithValue("@UnloadId", unloadId);
        saveMeth.Parameters.AddWithValue("@SupervisorId", supervisorId);
        saveMeth.Parameters.AddWithValue("@LooseBagCount", 3);
        saveMeth.Parameters.AddWithValue("@ReturnBagCount", 4);
        saveMeth.Parameters.AddWithValue("@Rows", $"[{{\"BagTypeId\":{bagTypeId},\"BagCount\":12,\"PerBagPrice\":3.50}}]");
        await saveMeth.ExecuteNonQueryAsync();

        await using var verifyMeth = new SqlCommand("SELECT COUNT(1), SUM(BagCount), SUM(TotalAmount) FROM dbo.t_SupervisorMethEntry WHERE UnloadId=@UnloadId", connection, methTransaction);
        verifyMeth.Parameters.AddWithValue("@UnloadId", unloadId);
        await using var methReader = await verifyMeth.ExecuteReaderAsync();
        await methReader.ReadAsync();
        Check(methReader.GetInt32(0) == 1 && methReader.GetInt32(1) == 12 && methReader.GetDecimal(2) == 42m, "supervisor Meth entry saves bag count and calculated amount correctly");
        await methReader.CloseAsync();

        await using var verifyLoose = new SqlCommand("SELECT LooseBagCount, ReturnBagCount FROM dbo.t_SupervisorMethLooseEntry WHERE UnloadId=@UnloadId", connection, methTransaction);
        verifyLoose.Parameters.AddWithValue("@UnloadId", unloadId);
        await using var extraReader = await verifyLoose.ExecuteReaderAsync();
        await extraReader.ReadAsync();
        Check(extraReader.GetInt32(0) == 3, "supervisor Meth entry saves loose bags separately without pricing");
        Check(extraReader.GetInt32(1) == 4, "supervisor Meth entry saves unloading-time return bags separately");
        await extraReader.CloseAsync();
    }
    finally
    {
        await methTransaction.RollbackAsync();
    }
}

Console.WriteLine($"ALL CURRENT WORKFLOW TESTS PASSED ({checks} checks)");
